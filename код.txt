inline std::string mv(int r, int c){
    return "\033["+std::to_string(r)+";"+std::to_string(c)+"H";
}
#define CLR   "\033[2J\033[H"
#define HCUR  "\033[?25l"
#define SCUR  "\033[?25h"

// ── Псевдографика ────────────────────────────────────────────────────────────
#define HL "─"
#define VL "│"
#define TL "┌"
#define TR "┐"
#define BL "└"
#define BR "┘"
#define LJ "├"
#define RJ "┤"

// ── Терминал ─────────────────────────────────────────────────────────────────
static struct termios _orig;
void rawOn()  { struct termios t; tcgetattr(0,&_orig); t=_orig; t.c_lflag&=~(ECHO|ICANON); t.c_cc[VMIN]=1; tcsetattr(0,TCSAFLUSH,&t); }
void rawOff() { tcsetattr(0,TCSAFLUSH,&_orig); }
void tsize(int&r,int&c){ struct winsize w; ioctl(1,TIOCGWINSZ,&w); r=w.ws_row?w.ws_row:40; c=w.ws_col?w.ws_col:120; }

enum Key { KENTER='\n', KCR='\r', KESC=27, KBS=127, KUP=256, KDN, KLT, KRT, KTAB='\t' };
int getch(){
    unsigned char c; if(read(0,&c,1)!=1)return -1;
    if(c==27){ unsigned char s[2]; read(0,&s[0],1); read(0,&s[1],1);
        if(s[0]=='['){
            if(s[1]=='A') return KUP;
            if(s[1]=='B') return KDN;
            if(s[1]=='C') return KRT;
            if(s[1]=='D') return KLT;
        }
        return KESC;
    }
    return c;
}

// ── Вспомогательные функции вывода ───────────────────────────────────────────
void at(int r,int c){ std::cout<<mv(r,c); }

void box(int r,int c,int h,int w,const char*clr=""){
    std::string line; for(int i=0;i<w-2;i++)line+=HL;
    std::cout<<clr; at(r,c); std::cout<<TL<<line<<TR;
    for(int i=1;i<h-1;i++){ at(r+i,c);std::cout<<VL; at(r+i,c+w-1);std::cout<<VL; }
    at(r+h-1,c); std::cout<<BL<<line<<BR; if(*clr)std::cout<<RST;
}

void fill(int r,int c,int h,int w,const char*bg){
    std::string s(w-2,' ');
    for(int i=1;i<h-1;i++){ at(r+i,c+1); std::cout<<bg<<s<<RST; }
}

void pr(int r,int c,const std::string&t,const char*clr=""){
    at(r,c); if(*clr)std::cout<<clr; std::cout<<t; if(*clr)std::cout<<RST;
}

void hdiv(int r,int c,int w){
    std::string s; s+=LJ; for(int i=0;i<w-2;i++)s+=HL; s+=RJ;
    at(r,c); std::cout<<BBLK<<s<<RST;
}

void btn(int r,int c,const std::string&lbl,bool act=true){
    at(r,c);
    if(act) std::cout<<BGBLU<<BWHT<<BOLD<<" "<<lbl<<" "<<RST;
    else    std::cout<<BGDK <<BWHT     <<" "<<lbl<<" "<<RST;
}

void field(int r,int c,int w,const std::string&val,bool focus,bool pwd=false){
    std::string d=pwd?std::string(val.size(),'*'):val;
    d.resize(w,' '); at(r,c);
    if(focus) std::cout<<"\033[47m"<<"\033[30m"<<d<<RST;
    else      std::cout<<BGDK<<BWHT<<d<<RST;
}

void append(std::string&s,int k,int max=48){
    if(k==KBS){ if(!s.empty())s.pop_back(); }
    else if(k>=32&&k<256&&(int)s.size()<max) s+=(char)k;
}

    Screen scr = LOGIN;
    std::string user = "";

// ── Боковое меню ─────────────────────────────────────────────────────────────
struct NI { const char*lbl; Screen scr; };
const NI NAV[]={{"Главная",DASH},{"Операции",OPS},{"Счета",ACCS},
                {"Отчёты",REPS},{"Справочники",REFS},
                {"Бюджеты",BUDGETS},{"Настройки",SETT}};
const int NLEN=7;

int sidebar(int rows, Screen cur){
    const int W=16;
    for(int r=2;r<=rows;r++){ at(r,1); std::cout<<BGDK<<std::string(W,' ')<<RST; }
    pr(2,2, BCYN+std::string(BOLD)+"УКДДС"+RST, "");
    for(int i=0;i<NLEN;i++){
        at(4+i,1);
        std::string lbl=" "; lbl+=NAV[i].lbl; lbl.resize(W,' ');
        if(NAV[i].scr==cur) std::cout<<BGBLU<<BWHT<<BOLD<<lbl<<RST;
        else                 std::cout<<BGDK <<BWHT     <<lbl<<RST;
    }
    at(rows,1); std::cout<<BGDK<<BWHT<<" Выход"<<std::string(W-7,' ')<<RST;
    return W;
}

void header(int cols,const char*title,const std::string&user){
    std::string bar(cols,' ');
    at(1,1); std::cout<<BGBLU<<BWHT<<BOLD<<bar<<RST;
    pr(1,2, std::string(BGBLU)+BWHT+BOLD+" УКДДС │ "+title+RST,"");
    std::string info=" "+user+" ";
    at(1,cols-(int)info.size()); std::cout<<BGBLU<<BWHT<<info<<RST;
}

// ── ОБРАБОТКА ВВОДА ──────────────────────────────────────────────────────────
void handleLogin(App&a, int k){
    switch(k){
        case KTAB:  a.lFld=(a.lFld+1)%2; break;
        case KENTER: case KCR:
            a.user=a.lUser.empty()?"Пользователь":a.lUser;
            a.scr=DASH; a.msg="";
            break;
        case 'q': case 'Q': case KESC: a.run=false; break;
        default:
            if(a.lFld==0) append(a.lUser,k);
            else          append(a.lPass,k);
            a.msg="";
    }
}

void handleMain(App&a, int k){
    if(k=='q'||k=='Q')  { a.run=false; return; }
    if(k==KESC)         { a.scr=LOGIN; return; }
    if(k=='1') a.scr=DASH;
    if(k=='2') a.scr=OPS;
    if(k=='3') a.scr=ACCS;
    if(k=='4') a.scr=REPS;
    if(k=='5') a.scr=REFS;
    if(k=='6') a.scr=BUDGETS;
    if(k=='7') a.scr=SETT;
    if((k=='n'||k=='N')&&(a.scr==OPS||a.scr==DASH)) a.scr=NEWOP;
    if(k==KLT&&a.scr==OPS&&a.opPage>1) a.opPage--;
    if(k==KRT&&a.scr==OPS&&a.opPage<2) a.opPage++;
    if(k==KUP&&a.scr==BUDGETS&&a.budRow>0) a.budRow--;
    if(k==KDN&&a.scr==BUDGETS&&a.budRow<3) a.budRow++;
}

// ── MAIN ─────────────────────────────────────────────────────────────────────
int main(){
    int rows,cols; tsize(rows,cols);
    if(rows<24) rows=24;
    if(cols<80) cols=80;
    rawOn();
    std::cout<<HCUR;

    App a;
    while(a.run){
        tsize(rows,cols);
        if(rows<24) rows=24;
        if(cols<80) cols=80;
        switch(a.scr){
            case LOGIN:   showLogin(a,rows,cols);   break;
            case DASH:    showDash(a,rows,cols);    break;
            case OPS:     showOps(a,rows,cols);     break;
            case ACCS:    showAccs(a,rows,cols);    break;
            case REPS:    showReps(a,rows,cols);    break;
            case REFS:    showRefs(a,rows,cols);    break;
            case NEWOP:   showNewOp(a,rows,cols);   break;
            case BUDGETS: showBudgets(a,rows,cols); break;
            case SETT:    showSett(a,rows,cols);    break;
        }
        std::cout.flush();
        int k=getch();
        if(a.scr==LOGIN)         handleLogin(a,k);
        else if(a.scr==NEWOP){ if(k==KESC||k=='q'||k=='Q')a.scr=OPS; }
        else                     handleMain(a,k);
    }

    std::cout<<SCUR<<CLR;
    rawOff();
    std::cout<<"До свидания!\n";
    return 0;
}
